// Солнцекофе — payment relay // // Why this exists: our main bot runs on Cloudflare Workers, which cannot be // configured to trust custom root certificates. T-Bank's payment endpoint // (securepay.tinkoff.ru) now serves a certificate chain signed by the // Russian Ministry of Digital Development (Minsifry) root CA, which isn't // in most default trust stores — Cloudflare Workers' outbound fetch() has // no way to add it, so calls from there fail with a 526 error. This tiny // service does *only* the two things that need to talk to T-Bank directly: // starting a payment, and receiving T-Bank's confirmation webhook. It runs // as a normal Node.js process, where we *can* add the extra trusted CA. // // Everything else (bot messages, staff notifications, "Офис цветов" posts, // reminders, order accept/done flow) still lives entirely in worker.js on // Cloudflare — this service only ever talks to Firebase and Telegram to // finish the job once a payment is confirmed. const http = require('http'); const https = require('https'); const crypto = require('crypto'); const fs = require('fs'); const PORT = process.env.PORT || 3000; const TBANK_TERMINAL_KEY = process.env.TBANK_TERMINAL_KEY; const TBANK_PASSWORD = process.env.TBANK_PASSWORD; const DB_URL = process.env.DB_URL; const BOT_TOKEN = process.env.BOT_TOKEN; const OWNER_CHAT_ID = process.env.OWNER_CHAT_ID; const SHOP1_CHAT_ID = process.env.SHOP1_CHAT_ID; const SHOP2_CHAT_ID = process.env.SHOP2_CHAT_ID; const WEBAPP_URL = process.env.WEBAPP_URL || ''; const PUBLIC_BASE_URL = process.env.PUBLIC_BASE_URL || ''; const API = `https://api.telegram.org/bot${BOT_TOKEN}`; // ---------- Russian Trusted CA (for talking to T-Bank only) ---------- let tbankAgent = https.globalAgent; // fallback if certs are missing for any reason try { const caCerts = []; for (const f of ['russian_trusted_root_ca.pem', 'russian_trusted_sub_ca.pem']) { if (fs.existsSync(f)) caCerts.push(fs.readFileSync(f)); } if (caCerts.length) { tbankAgent = new https.Agent({ ca: caCerts.concat(undefined).filter(Boolean) }); } else { console.error('Russian CA certs not found on disk — Tinkoff calls will likely fail. Check fetch-certs.js ran.'); } } catch (e) { console.error('Failed to load Russian CA certs:', e.message); } // ---------- tiny fetch-like helper using Node's https, with a chosen agent ---------- function httpsJson(urlStr, { method = 'GET', body, agent } = {}) { return new Promise((resolve, reject) => { const u = new URL(urlStr); const data = body ? JSON.stringify(body) : null; const req = https.request({ hostname: u.hostname, path: u.pathname + u.search, method, agent: agent || https.globalAgent, headers: Object.assign( { 'Content-Type': 'application/json' }, data ? { 'Content-Length': Buffer.byteLength(data) } : {} ), }, (res) => { let raw = ''; res.on('data', (c) => (raw += c)); res.on('end', () => { let parsed = null; try { parsed = JSON.parse(raw); } catch (e) { /* leave null */ } resolve({ status: res.statusCode, raw, json: parsed }); }); }); req.on('error', reject); if (data) req.write(data); req.end(); }); } function fbGet(path) { return httpsJson(`${DB_URL}${path}.json`).then(r => r.json); } function fbPut(path, body) { return httpsJson(`${DB_URL}${path}.json`, { method: 'PUT', body }); } function fbPatch(path, body) { return httpsJson(`${DB_URL}${path}.json`, { method: 'PATCH', body }); } function fbPost(path, body) { return httpsJson(`${DB_URL}${path}.json`, { method: 'POST', body }); } function fbDelete(path) { return httpsJson(`${DB_URL}${path}.json`, { method: 'DELETE' }); } // ---------- Telegram helpers (mirrors worker.js) ---------- async function sendMessage(chatId, text) { const r = await httpsJson(`${API}/sendMessage`, { method: 'POST', body: { chat_id: chatId, text, parse_mode: 'HTML' } }); return !!(r.json && r.json.ok); } async function sendMessageGetId(chatId, text) { const r = await httpsJson(`${API}/sendMessage`, { method: 'POST', body: { chat_id: chatId, text, parse_mode: 'HTML' } }); return r.json && r.json.ok ? r.json.result.message_id : null; } async function pinMessage(chatId, messageId) { if (!messageId) return; try { await httpsJson(`${API}/pinChatMessage`, { method: 'POST', body: { chat_id: chatId, message_id: messageId, disable_notification: true } }); } catch (e) { /* bot may not be admin — non-fatal */ } } async function notifyOnShiftStaff(shopId, text) { try { const obj = await fbGet(`/staff/${shopId}`); if (!obj) return; for (const id in obj) { const m = obj[id]; if (m && m.tg && m.active !== false) await sendMessage(m.tg, text); } } catch (e) { /* non-fatal */ } } // ---------- prep time + loyalty (mirrors worker.js) ---------- const PREP_MINUTES = { pizza: 15, default: 6 }; function prepMinutesForItems(items) { if (!Array.isArray(items)) return 6; return items.reduce((sum, it) => sum + (PREP_MINUTES[it.category] || PREP_MINUTES.default) * (it.qty || 1), 0); } const LOYALTY_GOAL = 7; async function awardLoyaltyStamps(guestId, count) { try { const data = (await fbGet(`/guests/${guestId}`)) || {}; const isFirstEverStamp = !data.stamps || data.stamps === 0; const total = (data.stamps || 0) + count; const justRewarded = total >= LOYALTY_GOAL; const stamps = total % LOYALTY_GOAL; await fbPut(`/guests/${guestId}`, Object.assign({}, data, { stamps, justRewarded, lastStampAt: Date.now(), reminded: false })); if (isFirstEverStamp && data.referredBy && !data.referralRewarded) { const refData = (await fbGet(`/guests/${data.referredBy}`)) || {}; const refTotal = (refData.stamps || 0) + 1; const refJustRewarded = refTotal >= LOYALTY_GOAL; const refStamps = refTotal % LOYALTY_GOAL; await fbPut(`/guests/${data.referredBy}`, Object.assign({}, refData, { stamps: refStamps, justRewarded: refJustRewarded, lastStampAt: Date.now(), reminded: false })); await fbPatch(`/guests/${guestId}`, { referralRewarded: true }); await fbPut(`/guests/${data.referredBy}/referrals/${guestId}`, { at: Date.now() }); } } catch (e) { console.error('awardLoyaltyStamps failed', e.message); } } // ---------- T-Bank token signing (identical algorithm to worker.js) ---------- function buildTbankToken(rootParams, password) { const withPassword = Object.assign({}, rootParams, { Password: password }); const keys = Object.keys(withPassword).sort(); const concat = keys.map((k) => String(withPassword[k])).join(''); return crypto.createHash('sha256').update(concat).digest('hex'); } async function callTbankInitWithRetry(initPayload) { const attempts = 3; let lastError = null; for (let i = 0; i < attempts; i++) { try { const res = await httpsJson('https://securepay.tinkoff.ru/v2/Init', { method: 'POST', body: initPayload, agent: tbankAgent }); if (res.json) return { data: res.json, error: null }; lastError = `non-JSON response, status ${res.status}: ${res.raw.slice(0, 300)}`; console.error(`Tbank Init attempt ${i + 1}/${attempts}:`, lastError); } catch (e) { lastError = e.message; console.error(`Tbank Init attempt ${i + 1}/${attempts}: network error:`, e.message); } if (i < attempts - 1) await new Promise((r) => setTimeout(r, 400 + i * 400)); } return { data: null, error: lastError }; } // ---------- /create-payment ---------- async function handleCreatePayment(body, res) { const reply = (obj) => { res.writeHead(200, { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }); res.end(JSON.stringify(obj)); }; if (!TBANK_TERMINAL_KEY || !TBANK_PASSWORD) { return reply({ ok: false, error: 'Оплата картой пока не настроена (нет ключей терминала).' }); } const orderId = 'sk' + Date.now() + Math.floor(Math.random() * 1000); const amountKopecks = Math.round((body.total || 0) * 100); const webAppBase = WEBAPP_URL.split('?')[0]; const rootParams = { TerminalKey: TBANK_TERMINAL_KEY, Amount: amountKopecks, OrderId: orderId, Description: `Заказ в Солнцекофе №${body.number}`, NotificationURL: `${PUBLIC_BASE_URL}/tbank-webhook`, SuccessURL: `${webAppBase}?paid=${orderId}`, FailURL: `${webAppBase}?failed=1`, }; const token = buildTbankToken(rootParams, TBANK_PASSWORD); const receiptItems = (body.items || []).map((it) => ({ Name: String(it.name || 'Позиция меню').slice(0, 128), Price: Math.round((it.price || 0) * 100), Quantity: it.qty || 1, Amount: Math.round((it.price || 0) * 100 * (it.qty || 1)), Tax: 'none', PaymentMethod: 'full_payment', PaymentObject: 'commodity', MeasurementUnit: 'шт', })); const contact = (body.receiptContact || '').trim(); const isEmail = contact.includes('@'); const receipt = { Taxation: 'usn_income', Items: receiptItems }; if (isEmail) receipt.Email = contact; else receipt.Phone = contact; const initPayload = Object.assign({}, rootParams, { Token: token, Receipt: receipt }); const { data, error } = await callTbankInitWithRetry(initPayload); if (!data) return reply({ ok: false, error: 'Не удалось связаться с банком: ' + error }); if (!data.Success) return reply({ ok: false, error: data.Message || data.Details || 'Платёж не создан.' }); try { // Stash PaymentId alongside the pending order so it can be attached to the // real order once confirmed — needed later to issue a refund. await fbPut(`/pending_payments/${orderId}`, Object.assign({}, body, { tbankPaymentId: data.PaymentId })); } catch (e) { /* webhook just won't find it */ } reply({ ok: true, paymentUrl: data.PaymentURL, orderId }); } // ---------- /refund-payment ---------- // Full refund only (no partial-refund UI yet) — matches T-Bank's Test 8 // scenario: call Cancel with just PaymentId, no Amount/Receipt needed. async function handleRefundPayment(body, res) { const reply = (obj) => { res.writeHead(200, { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }); res.end(JSON.stringify(obj)); }; if (!TBANK_TERMINAL_KEY || !TBANK_PASSWORD) { return reply({ ok: false, error: 'Оплата картой не настроена — возврат невозможен.' }); } const paymentId = body.paymentId; if (!paymentId) return reply({ ok: false, error: 'Не найден PaymentId этого заказа — возврат невозможен.' }); const rootParams = { TerminalKey: TBANK_TERMINAL_KEY, PaymentId: String(paymentId) }; const token = buildTbankToken(rootParams, TBANK_PASSWORD); const cancelPayload = Object.assign({}, rootParams, { Token: token }); let data; try { const r = await httpsJson('https://securepay.tinkoff.ru/v2/Cancel', { method: 'POST', body: cancelPayload, agent: tbankAgent }); data = r.json; } catch (e) { return reply({ ok: false, error: 'Не удалось связаться с банком: ' + e.message }); } if (!data) return reply({ ok: false, error: 'Банк ответил не тем, что мы ожидали.' }); if (!data.Success) return reply({ ok: false, error: data.Message || data.Details || 'Возврат не выполнен.' }); if (body.orderId) { try { await fbPatch(`/orders/${body.orderId}`, { refunded: true, refundedAt: Date.now(), refundStatus: data.Status }); } catch (e) { /* refund still went through at the bank even if this write fails */ } } reply({ ok: true, status: data.Status }); } // ---------- /tbank-webhook ---------- async function handleTbankWebhook(body, res) { res.writeHead(200, { 'Content-Type': 'text/plain' }); const { Token } = body; const scalarFields = {}; for (const k in body) if (k !== 'Token' && typeof body[k] !== 'object') scalarFields[k] = body[k]; const expectedToken = buildTbankToken(scalarFields, TBANK_PASSWORD); if (expectedToken !== Token) { console.error('Tbank webhook: token mismatch, ignoring.'); return res.end('OK'); } if (body.Status === 'CONFIRMED') { const orderId = body.OrderId; try { const pending = await fbGet(`/pending_payments/${orderId}`); if (pending) { const shopChatMap = { 1: SHOP1_CHAT_ID, 2: SHOP2_CHAT_ID }; const targetChatId = shopChatMap[pending.shopId] || OWNER_CHAT_ID; const prepMinutes = prepMinutesForItems(pending.items); const text = `🔔🔔🔔 НОВЫЙ ЗАКАЗ №${pending.number}\n` + `📍 ${pending.shopName}\n\n${pending.itemsText}\n\n` + `Итого: ${pending.total} ₽\n` + `Оплата: Онлайн (Т-Касса) ✅\n` + `Забрать: ${pending.pickup}\n` + `Время на приготовление: ~${prepMinutes} мин (пойдёт с момента, как примут заказ)\n` + `Имя: ${pending.guestName}\nID гостя: ${pending.clientId}`; const msgId = await sendMessageGetId(targetChatId, text); await pinMessage(targetChatId, msgId); if (OWNER_CHAT_ID && targetChatId !== OWNER_CHAT_ID) await sendMessage(OWNER_CHAT_ID, `(копия) ${text}`); await notifyOnShiftStaff(pending.shopId, text); const createRes = await fbPost('/orders', { number: pending.number, shopId: pending.shopId, shopName: pending.shopName, shopAddr: pending.shopAddr, itemsText: pending.itemsText, itemsCount: pending.itemsCount, total: pending.total, tip: pending.tip || 0, guestName: pending.guestName, payMethod: 'Онлайн (Т-Касса)', pickup: pending.pickup, clientId: pending.clientId, status: 'new', createdAt: Date.now(), prepMinutes, paid: true, tbankPaymentId: pending.tbankPaymentId, }); const newId = createRes.json && createRes.json.name; await fbPut(`/order_lookup/${orderId}`, newId); await fbDelete(`/pending_payments/${orderId}`); const coffeeQty = (pending.items || []).filter((it) => it.category === 'coffee').reduce((s, it) => s + (it.qty || 1), 0); if (coffeeQty > 0 && pending.clientId) await awardLoyaltyStamps(pending.clientId, coffeeQty); } } catch (e) { console.error('Tbank webhook processing failed', e.message); } } res.end('OK'); } // ---------- tiny HTTP server ---------- const server = http.createServer((req, res) => { if (req.method === 'OPTIONS') { res.writeHead(204, { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', }); return res.end(); } if (req.method === 'GET') { res.writeHead(200, { 'Content-Type': 'text/plain; charset=utf-8' }); return res.end('Солнцекофе payment relay работает ☀️'); } let raw = ''; req.on('data', (c) => (raw += c)); req.on('end', async () => { let body = {}; try { body = JSON.parse(raw || '{}'); } catch (e) { /* keep {} */ } if (req.url === '/create-payment') return handleCreatePayment(body, res); if (req.url === '/tbank-webhook') return handleTbankWebhook(body, res); if (req.url === '/refund-payment') return handleRefundPayment(body, res); res.writeHead(404, { 'Content-Type': 'text/plain' }); res.end('not found'); }); }); server.listen(PORT, () => console.log('Payment relay listening on', PORT));
